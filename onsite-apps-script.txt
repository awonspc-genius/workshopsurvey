   ▶ 이 스크립트가 하는 일
   · 사진   → 내 드라이브 > 'BR ONSITE 매장리포트 사진' 폴더에 JPG로 저장
   · 답변   → 이 시트의 'OnSite' 탭에 한 줄로 기록
   · 사진 표시 → 시트 뒤쪽 6개 컬럼에 =IMAGE() 수식이 들어가 시트 안에서 바로 보임
   설문 스크립트와는 완전히 분리돼 있어서 설문 수집에 영향을 주지 않습니다.

   ▶ 배포 (5분, 한 번만)

        설명       : ONSITE
        실행 사용자 : 나
        액세스 권한 : 링크가 있는 모든 사용자  (또는 회사 도메인 내 모든 사용자)
      → 배포 → 권한 승인
      ※ 이 시트는 새 시트라 "새 배포"가 맞습니다. 설문 스크립트와 별개예요.
   4. 나온 웹 앱 URL(https://script.google.com/macros/s/.../exec)을 복사해서
      br-store-report.html 상단
        var UPLOAD_ENDPOINT = "";
      안에 붙여넣고 다시 올리면 끝입니다.

var ONSITE_NAME = 'OnSite';                    // 기록이 쌓이는 탭
var FOLDER_NAME = 'BR ONSITE 매장리포트 사진';   // 사진 폴더
var ONSITE_HEADERS = ['sentAt','id','store','date','author','hour','oneline','survey','notes','photos',
                      '① 전경','② 외창','③ 메뉴보드','④ 디핑구','⑤ MD장','⑥ 포스'];
var IMG_COL_START  = 11;   // '① 전경' 컬럼 (1부터 셈)
var IMG_COUNT      = 6;
var IMG_W          = 150;  // 시트에 표시할 썸네일 크기(px)
var IMG_H          = 105;

function getOnsiteSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(ONSITE_NAME);
  if (!sh) {
    sh = ss.insertSheet(ONSITE_NAME);
    sh.getRange(1, 1, 1, ONSITE_HEADERS.length).setValues([ONSITE_HEADERS]).setFontWeight('bold');
    sh.setFrozenRows(1);
    for (var c = 0; c < IMG_COUNT; c++) sh.setColumnWidth(IMG_COL_START + c, IMG_W + 10);
  }
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, ONSITE_HEADERS.length).setValues([ONSITE_HEADERS]).setFontWeight('bold');
  }
  return sh;
}

function getFolder_(name, parent) {
  var it = parent ? parent.getFoldersByName(name) : DriveApp.getFoldersByName(name);
  if (it.hasNext()) return it.next();
  return parent ? parent.createFolder(name) : DriveApp.createFolder(name);
}

/* ── 저장 ─────────────────────────────────────────────── */
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(30000);
  try {
    var body = JSON.parse(e.postData.contents);
    /* ---- 매장 리포트 저장  ---- */
    if (body.type === 'onsite') {
      var store = String(body.store || '무제').replace(/[\\\/:*?"<>|]/g, '_');
      var date  = String(body.date || '').slice(0, 10);
      var folder = getFolder_(date + ' ' + store, getFolder_(FOLDER_NAME));

      var urls = [];
      (body.photos || []).forEach(function (ph, i) {
        var parts = String(ph.data).split(',');
        var bytes = Utilities.base64Decode(parts[1]);
        var name  = date + '_' + store + '_' + (i + 1) + '_' + (ph.slot || 'photo') + '.jpg';
        var old = folder.getFilesByName(name);
        while (old.hasNext()) old.next().setTrashed(true);   // 재저장이면 교체
        var file = folder.createFile(Utilities.newBlob(bytes, 'image/jpeg', name));
        try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (err) {}
        urls.push('https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w600');
      });

      var sv = body.survey || {};
      var orow = [
        new Date().toISOString(), body.id || '', body.store || '', body.date || '',
        sv.author || '', sv.hour || '', sv.oneline || '',
        JSON.stringify(sv), JSON.stringify(body.notes || {}), JSON.stringify(urls)
      ];

      var osh = getOnsiteSheet_();
      var ids = osh.getLastRow() > 1
        ? osh.getRange(2, 2, osh.getLastRow() - 1, 1).getValues().map(function (x) { return x[0]; })
        : [];
      var at = ids.indexOf(body.id);
      var rowNum;
      if (at > -1) { rowNum = at + 2; osh.getRange(rowNum, 1, 1, orow.length).setValues([orow]); }
      else { osh.appendRow(orow); rowNum = osh.getLastRow(); }

      /* 시트 안에서 사진이 바로 보이도록 IMAGE 수식을 넣습니다.
         (파일은 드라이브에 있고 시트에는 수식만 — 용량은 늘지 않습니다) */
      for (var c = 0; c < IMG_COUNT; c++) {
        var cell = osh.getRange(rowNum, IMG_COL_START + c);
        if (urls[c]) cell.setFormula('=IMAGE("' + urls[c] + '", 4, ' + IMG_H + ', ' + IMG_W + ')');
        else cell.clearContent();
      }
      osh.setRowHeight(rowNum, IMG_H + 10);

      return ContentService.createTextOutput('ok');
    }

    return ContentService.createTextOutput('ignored');
  } catch (err) {
    return ContentService.createTextOutput('error: ' + err);
  } finally {
    lock.releaseLock();
  }
}

/* ── 읽기 (JSONP) ─────────────────────────────────────── */
function doGet(e) {
  var action = (e.parameter && e.parameter.action) || '';
  var cb = (e.parameter && e.parameter.callback) || 'cb';
  var out;

  /* ---- 매장 리포트 목록  ---- */
  if (action === 'onsite_list') {
    var osh = getOnsiteSheet_();
    var rows = [];
    if (osh.getLastRow() > 1) {
      var vals = osh.getRange(2, 1, osh.getLastRow() - 1, 10).getValues();   // 앞 10개 컬럼만 (뒤는 이미지 수식)
      rows = vals.map(function (v) {
        var photos = [];
        try { photos = JSON.parse(v[9] || '[]'); } catch (err) {}
        return {
          sentAt: v[0] instanceof Date ? v[0].toISOString() : String(v[0]),
          id: v[1], store: v[2],
          date: v[3] instanceof Date ? Utilities.formatDate(v[3], 'Asia/Seoul', 'yyyy-MM-dd') : String(v[3]),
          author: v[4], hour: v[5], oneline: v[6], photos: photos
        };
      });
    }
    out = { rows: rows };

  } else {
    out = { ok: true };
  }

  return ContentService
    .createTextOutput(cb + '(' + JSON.stringify(out) + ')')
    .setMimeType(ContentService.MimeType.JAVASCRIPT);
}

/* ============================================================
   ▶ 알아두실 점

   · 시트의 '① 전경' ~ '⑥ 포스' 컬럼에는 =IMAGE() 수식이 들어가서
     시트 안에서 사진이 바로 보입니다. 파일 자체는 드라이브에 있고
     시트에는 수식만 있어서 용량은 늘지 않아요. 사진이 회색 상자로
     보이면 잠시 후 새로고침해 보세요(구글이 썸네일을 만드는 중입니다).

   · 매장 사진은 "링크가 있는 모든 사용자 보기"로 공유됩니다.
     페이지와 시트에서 썸네일을 보여주려면 필요한 설정이에요. 걱정되면
     setSharing 줄을 지우세요(대신 썸네일이 안 보입니다).

   · 촬영 시 인물 얼굴이 들어가지 않도록 하는 안내를 지켰는지
     확인해 주세요. 드라이브에 남는 자료입니다.

   · 코드를 고친 뒤에는 "배포 관리 → 편집(연필) → 새 버전 → 배포"로
     기존 배포의 버전을 올리세요. 그래야 주소가 그대로 유지됩니다.
   ============================================================ */
