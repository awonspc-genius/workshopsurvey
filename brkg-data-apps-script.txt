/*  BRK Genius 공용 데이터 저장소 — Google Apps Script
 *  IP 업무(후보·상품·자료)와 뮤직쇼 사진을 한 시트에 모아 둡니다.
 *  한 건이 한 줄로 들어가므로 자료가 늘어도 칸 용량 제한에 걸리지 않습니다.
 *
 *  ── 처음 한 번만 하면 되는 설치 순서 ────────────────────────────
 *  1. 구글 드라이브에서 새 스프레드시트를 만듭니다. 이름은 아무거나.
 *     (이미 쓰는 시트에 합치려면 새로 만들지 말고, 아래 SHEET_ID 설명을 보세요)
 *  2. 메뉴 [확장 프로그램] > [Apps Script] 를 엽니다.
 *  3. 기본으로 있는 코드를 전부 지우고 이 파일 내용을 그대로 붙여넣습니다.
 *  4. 저장한 뒤 [배포] > [새 배포] 를 누릅니다.
 *       유형        : 웹 앱
 *       실행 사용자 : 나
 *       액세스 권한 : 모든 사용자
 *  5. 배포하면 나오는 https://script.google.com/macros/s/....../exec 주소를 복사합니다.
 *  6. 그 주소를 아래 세 곳의 DATA_URL 에 붙여넣으면 끝입니다.
 *       · ip/index.html          (var DATA_URL = "";)
 *       · ip-contract/index.html (var DATA_URL = "";)
 *       · feed/index.html        (var DATA_URL = "";)
 *     따옴표 안에 붙여넣습니다. 예) var DATA_URL = "https://script.google.com/macros/s/AKfy.../exec";
 *
 *  ── 뮤직쇼 사진 ────────────────────────────────────────────────
 *  사진은 시트가 아니라 구글 드라이브에 올라갑니다.
 *  처음 올릴 때 내 드라이브에 'BRK 뮤직쇼 사진' 폴더가 자동으로 만들어지고,
 *  올린 사진은 '링크가 있는 사람은 누구나 보기'로 공개됩니다.
 *  시트에는 사진 주소만 남으므로 칸 용량 제한에 걸리지 않습니다.
 *  (처음 한 번은 드라이브 접근 권한을 묻는 창이 뜹니다. 허용해 주세요.)
 *
 *  ── 이미 쓰는 시트에 합치기 ────────────────────────────────────
 *  인스타 피드 플래닝처럼 이미 돌아가는 시트가 있으면, 그 시트에
 *  탭만 더 붙이는 편이 낫습니다. 자료가 한 파일에 모입니다.
 *  주의: 그 시트에 붙어 있는 Apps Script 안에 이 코드를 같이 넣으면 안 됩니다.
 *        doGet / doPost 가 두 개가 되어 둘 다 망가집니다.
 *  대신 이렇게 하세요.
 *    1) script.google.com 에서 [새 프로젝트] 를 만듭니다. (시트에서 열지 않습니다)
 *    2) 이 코드를 붙여넣습니다.
 *    3) 아래 SHEET_ID 에 그 시트의 ID 를 넣습니다.
 *    4) 웹 앱으로 배포합니다. 주소는 기존 피드 것과 따로 생깁니다.
 *  이러면 피드 스크립트는 그대로 두고, 같은 시트 파일에
 *  ip · md · docs · music · ipmaster · meta 탭이 더 생깁니다.
 *
 *  ── 코드를 고친 뒤에는 ──────────────────────────────────────────
 *  [배포] > [배포 관리] > 연필 아이콘 > 버전 '새 버전' > [배포]
 *  (새 배포를 또 만들면 주소가 바뀌니 반드시 '배포 관리'에서 고치세요)
 */

/*  이미 쓰고 있는 시트(예: 인스타 피드 플래닝)에 탭만 더 붙이고 싶다면
 *  그 시트 주소의 가운데 긴 글자를 아래 따옴표 안에 넣으세요.
 *    https://docs.google.com/spreadsheets/d/[여기가 ID]/edit
 *  비워 두면 이 스크립트가 붙어 있는 시트에 만듭니다. */
var SHEET_ID = "";

var PHOTO_FOLDER = "BRK 뮤직쇼 사진";

var TABLES = {
  ip:    ["id","name","cat","stage","grp","slot","period","owner","contact","season","fee",
          "excl","mg","roy","rates","dur","img","temp","memo","created","updated","by"],
  md:    ["id","ipId","name","stage","qty","cost","price","memo","updated","by"],
  docs:  ["id","ipId","type","title","url","memo","updated","by"],
  music: ["id","date","title","place","memo","photos","updated","by"],
  ipmaster: ["id","grp","ip","kind","slot","agent","period","mg","roy","prod","dur","poc","excl","stage","memo","updated"],
  meta:     ["id","by","at","n","updated"]
};

function sheetFor(table) {
  var cols = TABLES[table];
  if (!cols) return null;
  var ss = SHEET_ID ? SpreadsheetApp.openById(SHEET_ID)
                    : SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(table);
  if (!sh) {
    sh = ss.insertSheet(table);
    sh.getRange(1, 1, 1, cols.length).setValues([cols]);
    sh.setFrozenRows(1);
  }
  return sh;
}

/* 칸에 넣을 수 있는 값으로 바꾼다. 객체·배열은 글자로 접어 넣는다. */
function pack(v) {
  if (v === undefined || v === null) return "";
  if (typeof v === "object") {
    try { return JSON.stringify(v); } catch (e) { return ""; }
  }
  return v;
}
/* 접어 넣었던 값을 되돌린다 */
function unpack(v) {
  if (typeof v !== "string") return v;
  var t = v.trim();
  if (t.length > 1 && (t.charAt(0) === "{" || t.charAt(0) === "[")) {
    try { return JSON.parse(t); } catch (e) { return v; }
  }
  return v;
}

function readAll(table) {
  var cols = TABLES[table], sh = sheetFor(table);
  var last = sh.getLastRow();
  if (last < 2) return [];
  var vals = sh.getRange(2, 1, last - 1, cols.length).getValues();
  var out = [];
  for (var i = 0; i < vals.length; i++) {
    if (!vals[i][0]) continue;                 // id 없는 줄은 지워진 것으로 본다
    var o = {};
    for (var c = 0; c < cols.length; c++) o[cols[c]] = unpack(vals[i][c]);
    if (table === "music" && !(o.photos instanceof Array)) o.photos = [];
    out.push(o);
  }
  return out;
}

function newId(table) {
  return table + "_" + Date.now() + "_" + Math.floor(Math.random() * 100000);
}

/* 한 건만 저장할 때 */
function upsert(table, rec) {
  return upsertMany(table, [rec])[0];
}

/*  여러 건을 한 번에 저장한다.
 *  예전에는 한 줄 쓸 때마다 시트를 다시 읽어서 89건이면 백 번 넘게 왕복했다.
 *  지금은 ID를 한 번만 읽고, 고칠 줄은 이어진 덩어리로 묶어서 한 번에 쓴다.  */
function upsertMany(table, recs) {
  var cols = TABLES[table], sh = sheetFor(table);
  if (!recs || !recs.length) return [];

  var last = sh.getLastRow();
  var rowOf = {};                                  /* id → 줄 번호 */
  if (last >= 2) {
    var ids = sh.getRange(2, 1, last - 1, 1).getValues();
    for (var i = 0; i < ids.length; i++) {
      var v = String(ids[i][0]);
      if (v) rowOf[v] = i + 2;
    }
  }

  /*  대표 이미지 지킴이.
   *  사진이 없는 컴퓨터에서 저장을 눌러도 시트에 있는 사진을 지우지 않는다.
   *  (사진은 한 사람 브라우저에만 있을 수 있어서, 빈 값으로 덮이면 되살릴 길이 없다)  */
  var imgAt = cols.indexOf("img"), oldImg = {};
  if (imgAt >= 0 && last >= 2) {
    var col = sh.getRange(2, imgAt + 1, last - 1, 1).getValues();
    for (var q = 0; q < col.length; q++) oldImg[q + 2] = String(col[q][0] || "");
  }

  var now = new Date().toISOString();
  var edits = [], adds = [], outIds = [];

  for (var j = 0; j < recs.length; j++) {
    var rec = recs[j];
    if (!rec.id) rec.id = newId(table);
    rec.updated = now;
    var row = cols.map(function (c) { return pack(rec[c]); });
    outIds.push(rec.id);

    var key = String(rec.id);
    if (rowOf[key]) {
      var r0 = rowOf[key];
      if (imgAt >= 0) {
        var incoming = String(row[imgAt] || "");
        /* 보내온 사진이 비었으면 시트에 있던 것을 그대로 둔다 */
        if (!incoming && oldImg[r0]) row[imgAt] = oldImg[r0];
        /* 칸 한도(5만 자)를 넘는 사진은 시트가 거부하므로 있던 것을 지킨다 */
        if (String(row[imgAt] || "").length > 49000) row[imgAt] = oldImg[r0] || "";
      }
      edits.push({ r: r0, v: row });
    } else {
      if (imgAt >= 0 && String(row[imgAt] || "").length > 49000) row[imgAt] = "";
      rowOf[key] = -1;                             /* 같은 요청 안에서 중복 추가 막기 */
      adds.push(row);
    }
  }

  /* 고칠 줄 : 번호순으로 정렬한 뒤 이어진 것끼리 묶어서 한 번에 쓴다 */
  edits.sort(function (a, b) { return a.r - b.r; });
  var k = 0;
  while (k < edits.length) {
    var start = k, block = [edits[k].v];
    while (k + 1 < edits.length && edits[k + 1].r === edits[k].r + 1) {
      k++; block.push(edits[k].v);
    }
    sh.getRange(edits[start].r, 1, block.length, cols.length).setValues(block);
    k++;
  }

  /* 새로 넣을 줄 : 맨 아래에 한 번에 붙인다 */
  if (adds.length) {
    sh.getRange(sh.getLastRow() + 1, 1, adds.length, cols.length).setValues(adds);
  }

  SpreadsheetApp.flush();
  return outIds;
}

function remove(table, id) {
  var sh = sheetFor(table);
  var last = sh.getLastRow();
  if (last < 2) return false;
  var ids = sh.getRange(2, 1, last - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(id)) { sh.deleteRow(i + 2); return true; }
  }
  return false;
}

/* ── 읽기 : JSONP (브라우저 보안에 막히지 않음) ── */
function doGet(e) {
  var p = (e && e.parameter) || {};
  var out;
  try {
    if (p.action === "list") {
      out = { ok: true, table: p.table, rows: readAll(p.table) };
    } else if (p.action === "all") {
      var all = {};
      for (var t in TABLES) all[t] = readAll(t);
      out = { ok: true, data: all };
    } else if (p.action === "ping") {
      out = { ok: true, tables: Object.keys(TABLES) };
    } else if (!p.action) {
      /* 주소만 쳤을 때 : 잘 돌아가고 있는지 한눈에 보이도록 */
      var cnt = {};
      for (var t2 in TABLES) cnt[t2] = readAll(t2).length;
      out = { ok: true, msg: "BRK Genius 저장소가 정상 동작 중입니다.",
              시트: (SHEET_ID ? SHEET_ID : "이 스크립트가 붙어 있는 시트"),
              표별_건수: cnt };
    } else {
      out = { ok: false, error: "action 을 알 수 없습니다 (list / all / ping)" };
    }
  } catch (err) {
    out = { ok: false, error: String(err) };
  }
  var body = JSON.stringify(out);
  if (p.callback) {
    return ContentService.createTextOutput(p.callback + "(" + body + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JSON);
}

function photoFolder() {
  var it = DriveApp.getFoldersByName(PHOTO_FOLDER);
  return it.hasNext() ? it.next() : DriveApp.createFolder(PHOTO_FOLDER);
}

/* 사진 한 장을 드라이브에 올리고, 해당 공연 기록의 photos 에 주소를 덧붙인다 */
function uploadPhoto(b) {
  var raw = String(b.data || "");
  var comma = raw.indexOf(",");
  if (comma > -1) raw = raw.slice(comma + 1);           // data:image/...;base64, 잘라내기
  var bytes = Utilities.base64Decode(raw);
  var blob = Utilities.newBlob(bytes, b.mime || "image/jpeg", b.name || ("photo_" + Date.now() + ".jpg"));
  var file = photoFolder().createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  var url = "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w1400";

  var sh = sheetFor("music"), cols = TABLES.music;
  var last = sh.getLastRow();
  if (last >= 2) {
    var ids = sh.getRange(2, 1, last - 1, 1).getValues();
    for (var i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) !== String(b.showId)) continue;
      var cell = sh.getRange(i + 2, cols.indexOf("photos") + 1);
      var arr = [];
      try { arr = JSON.parse(cell.getValue() || "[]"); } catch (e) { arr = []; }
      arr.push(url);
      cell.setValue(JSON.stringify(arr));
      sh.getRange(i + 2, cols.indexOf("updated") + 1).setValue(new Date().toISOString());
      return url;
    }
  }
  return url;   /* 기록을 못 찾아도 파일은 남는다 */
}

/* IP 대표 이미지를 드라이브에 올리고 ip 표의 img 칸에 주소를 적는다 */
function uploadIpPhoto(b) {
  var raw = String(b.data || "");
  var comma = raw.indexOf(",");
  if (comma > -1) raw = raw.slice(comma + 1);
  var blob = Utilities.newBlob(Utilities.base64Decode(raw), b.mime || "image/jpeg",
                               b.name || ("ip_" + Date.now() + ".jpg"));
  var file = photoFolder().createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  var url = "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w800";

  var sh = sheetFor("ip"), cols = TABLES.ip, last = sh.getLastRow();
  if (last >= 2) {
    var ids = sh.getRange(2, 1, last - 1, 1).getValues();
    for (var i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) !== String(b.ipId)) continue;
      var c = cols.indexOf("img");
      if (c < 0) break;
      sh.getRange(i + 2, c + 1).setValue(url);
      sh.getRange(i + 2, cols.indexOf("updated") + 1).setValue(new Date().toISOString());
      break;
    }
  }
  return url;
}

/* ── 쓰기 : POST ── */
function doPost(e) {
  var out;
  try {
    var b = JSON.parse(e.postData.contents);
    var lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try {
      if (b.type === "save") {
        out = { ok: true, id: upsert(b.table, b.rec) };
      } else if (b.type === "saveMany") {
        out = { ok: true, ids: upsertMany(b.table, b.recs || []) };
      } else if (b.type === "delete") {
        out = { ok: remove(b.table, b.id) };
      } else if (b.type === "photo") {
        out = { ok: true, url: uploadPhoto(b) };
      } else if (b.type === "ipphoto") {
        out = { ok: true, url: uploadIpPhoto(b) };
      } else {
        out = { ok: false, error: "type 을 알 수 없습니다" };
      }
    } finally { lock.releaseLock(); }
  } catch (err) {
    out = { ok: false, error: String(err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}
