var PHOTO_FOLDER = "BRK 뮤직쇼 사진";

var TABLES = {
  ip:    ["id","name","cat","stage","owner","contact","memo","start","end","royalty","updated","by"],
  md:    ["id","ipId","name","stage","qty","cost","price","memo","updated","by"],
  docs:  ["id","ipId","type","title","url","memo","updated","by"],
  music: ["id","date","title","place","memo","photos","updated","by"],
  ipmaster: ["id","grp","ip","kind","slot","agent","period","mg","roy","prod","dur","poc","excl","stage","memo","updated"],
  meta:     ["id","by","at","n","updated"]
};

function sheetFor(table) {
  var cols = TABLES[table];
  if (!cols) return null;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(table);
  if (!sh) {
    sh = ss.insertSheet(table);
    sh.getRange(1, 1, 1, cols.length).setValues([cols]);
    sh.setFrozenRows(1);
  }
  return sh;
}

function readAll(table) {
  var cols = TABLES[table], sh = sheetFor(table);
  var last = sh.getLastRow();
  if (last < 2) return [];
  var vals = sh.getRange(2, 1, last - 1, cols.length).getValues();
  var out = [];
  for (var i = 0; i < vals.length; i++) {
    if (!vals[i][0]) continue;                 // id 없는 줄은 지워진 것으로 본다
    var o = {};
    for (var c = 0; c < cols.length; c++) o[cols[c]] = vals[i][c];
    if (table === "music" && o.photos) {
      try { o.photos = JSON.parse(o.photos); } catch (e) { o.photos = []; }
    }
    out.push(o);
  }
  return out;
}

function upsert(table, rec) {
  var cols = TABLES[table], sh = sheetFor(table);
  if (!rec.id) rec.id = table + "_" + Date.now() + "_" + Math.floor(Math.random() * 1000);
  rec.updated = new Date().toISOString();
  if (table === "music" && rec.photos && typeof rec.photos !== "string") {
    rec.photos = JSON.stringify(rec.photos);
  }
  var row = cols.map(function (c) { return rec[c] === undefined ? "" : rec[c]; });

  var last = sh.getLastRow();
  if (last >= 2) {
    var ids = sh.getRange(2, 1, last - 1, 1).getValues();
    for (var i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) === String(rec.id)) {
        sh.getRange(i + 2, 1, 1, cols.length).setValues([row]);
        return rec.id;
      }
    }
  }
  sh.appendRow(row);
  return rec.id;
}

function remove(table, id) {
  var sh = sheetFor(table);
  var last = sh.getLastRow();
  if (last < 2) return false;
  var ids = sh.getRange(2, 1, last - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(id)) { sh.deleteRow(i + 2); return true; }
  }
  return false;
}

/* ── 읽기 : JSONP (브라우저 보안에 막히지 않음) ── */
function doGet(e) {
  var p = (e && e.parameter) || {};
  var out;
  try {
    if (p.action === "list") {
      out = { ok: true, table: p.table, rows: readAll(p.table) };
    } else if (p.action === "all") {
      var all = {};
      for (var t in TABLES) all[t] = readAll(t);
      out = { ok: true, data: all };
    } else if (p.action === "ping") {
      out = { ok: true, tables: Object.keys(TABLES) };
    } else {
      out = { ok: false, error: "action 을 알 수 없습니다" };
    }
  } catch (err) {
    out = { ok: false, error: String(err) };
  }
  var body = JSON.stringify(out);
  if (p.callback) {
    return ContentService.createTextOutput(p.callback + "(" + body + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JSON);
}

function photoFolder() {
  var it = DriveApp.getFoldersByName(PHOTO_FOLDER);
  return it.hasNext() ? it.next() : DriveApp.createFolder(PHOTO_FOLDER);
}

/* 사진 한 장을 드라이브에 올리고, 해당 공연 기록의 photos 에 주소를 덧붙인다 */
function uploadPhoto(b) {
  var raw = String(b.data || "");
  var comma = raw.indexOf(",");
  if (comma > -1) raw = raw.slice(comma + 1);           // data:image/...;base64, 잘라내기
  var bytes = Utilities.base64Decode(raw);
  var blob = Utilities.newBlob(bytes, b.mime || "image/jpeg", b.name || ("photo_" + Date.now() + ".jpg"));
  var file = photoFolder().createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  var url = "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w1400";

  var sh = sheetFor("music"), cols = TABLES.music;
  var last = sh.getLastRow();
  if (last >= 2) {
    var ids = sh.getRange(2, 1, last - 1, 1).getValues();
    for (var i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) !== String(b.showId)) continue;
      var cell = sh.getRange(i + 2, cols.indexOf("photos") + 1);
      var arr = [];
      try { arr = JSON.parse(cell.getValue() || "[]"); } catch (e) { arr = []; }
      arr.push(url);
      cell.setValue(JSON.stringify(arr));
      sh.getRange(i + 2, cols.indexOf("updated") + 1).setValue(new Date().toISOString());
      return url;
    }
  }
  return url;   /* 기록을 못 찾아도 파일은 남는다 */
}

/* ── 쓰기 : POST ── */
function doPost(e) {
  var out;
  try {
    var b = JSON.parse(e.postData.contents);
    var lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try {
      if (b.type === "save") {
        out = { ok: true, id: upsert(b.table, b.rec) };
      } else if (b.type === "saveMany") {
        var ids = (b.recs || []).map(function (r) { return upsert(b.table, r); });
        out = { ok: true, ids: ids };
      } else if (b.type === "delete") {
        out = { ok: remove(b.table, b.id) };
      } else if (b.type === "photo") {
        out = { ok: true, url: uploadPhoto(b) };
      } else {
        out = { ok: false, error: "type 을 알 수 없습니다" };
      }
    } finally { lock.releaseLock(); }
  } catch (err) {
    out = { ok: false, error: String(err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}
